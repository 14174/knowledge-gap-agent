# 阶段一可信人工门禁修复实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 `subagent-driven-development`（推荐）或 `executing-plans` 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法跟踪进度。

**目标：** 建立绑定内容版本的真实人工终审记录，让正式冻结无法由手填状态绕过，同时收紧高风险样本语义、隔离破坏性测试并交付可精确核对的人审清单。

**架构：** 模型 `ReviewRecord`、人工 `HumanReviewRecord`、内容修改 `HumanRevisionRecord` 三类记录职责分离。冻结器以可信语料重建目标哈希和人工需求，人工记录只进入审计集；夹具构建器通过显式 `--workspace-root` 在隔离副本运行。

**技术栈：** Python 3.11、Pydantic v2、pytest、规范 JSON/JSONL、现有 `uv` 工具链。

- [ ] **步骤 1：把弱反例写成失败测试**

为 `outdated` 分别构造：旧主张无 `valid_until`、当前主张无 `valid_from`、时间相等、旧主张与研究池当前主张无直接冲突。为 `conflict` 构造：可见两端冲突，但研究池裁决只关联一端。

```python
def outdated_data(
    *, old_until: datetime | None, current_from: datetime | None
) -> tuple[BenchmarkCase, KnowledgeEnvironment, dict[str, CorpusChunk], dict[str, Claim]]:
    target, environment, chunks, claims = valid_data()
    target = make_case(category="outdated", need_research=True)
    chunks["visible"] = chunks["visible"].model_copy(update={"claim_ids": ("old",)})
    claims["old"] = make_claim().model_copy(update={
        "claim_id": "old",
        "evidence_chunk_ids": ("visible",),
        "valid_from": None,
        "valid_until": old_until,
        "conflicts_with": ("claim1",),
    })
    claims["claim1"] = claims["claim1"].model_copy(update={
        "valid_from": current_from,
        "valid_until": None,
        "conflicts_with": ("old",),
    })
    return target, environment, chunks, claims


def conflict_data(
    adjudicator_conflicts: tuple[str, ...],
) -> tuple[BenchmarkCase, KnowledgeEnvironment, dict[str, CorpusChunk], dict[str, Claim]]:
    target, environment, chunks, claims = valid_data()
    target = make_case(category="conflict", need_research=True)
    chunks["visible"] = chunks["visible"].model_copy(
        update={"claim_ids": ("left", "right")}
    )
    claims["left"] = make_claim().model_copy(update={
        "claim_id": "left",
        "evidence_chunk_ids": ("visible",),
        "conflicts_with": ("right", "claim1"),
    })
    claims["right"] = make_claim().model_copy(update={
        "claim_id": "right",
        "evidence_chunk_ids": ("visible",),
        "conflicts_with": ("left",),
    })
    claims["claim1"] = claims["claim1"].model_copy(
        update={"conflicts_with": adjudicator_conflicts}
    )
    return target, environment, chunks, claims


def test_outdated_requires_strict_temporal_handoff() -> None:
    target, environment, chunks, claims = outdated_data(
        old_until=datetime(2026, 1, 2, tzinfo=timezone.utc),
        current_from=datetime(2026, 1, 2, tzinfo=timezone.utc),
    )
    codes = {issue.code for issue in validate_case(target, environment, chunks, claims)}
    assert "outdated_evidence_missing" in codes


def test_conflict_adjudicator_must_link_both_visible_sides() -> None:
    target, environment, chunks, claims = conflict_data(adjudicator_conflicts=("left",))
    codes = {issue.code for issue in validate_case(target, environment, chunks, claims)}
    assert "conflict_evidence_missing" in codes
```

def parse_named_json_blocks(text: str, marker: str) -> dict[str, dict]:
    pattern = re.compile(
        rf"(?ms)^<!-- {marker}:(?P<key>[^ ]+) -->\n```json\n(?P<body>.*?)\n```$"
    )
    return {
        match["key"]: json.loads(match["body"])
        for match in pattern.finditer(text)
    }


def load_jsonl(path: Path) -> list[dict]:
    return [json.loads(line) for line in path.read_text(encoding="utf-8").splitlines()]


def build_expected_cards_from_fixtures() -> dict[str, dict]:
    queue = {
        row["case_id"]: row
        for row in load_jsonl(FIXTURES / "benchmark/human_review_queue.jsonl")
    }
    inputs = {
        row["case"]["case_id"]: row
        for row in load_jsonl(FIXTURES / "benchmark/review_inputs.jsonl")
    }
    reviews = {
        row["case_id"]: row
        for row in load_jsonl(FIXTURES / "benchmark/reviews.jsonl")
    }
    drafts = {
        row["case"]["case_id"]: row["case"]
        for row in load_jsonl(FIXTURES / "benchmark/drafts.jsonl")
    }
    result = {}
    for case_id, queued in queue.items():
        item = inputs[case_id]
        case = item["case"]
        environment = item["environment"]
        result[case_id] = {
            "case_id": case_id,
            "base_question_id": case["base_question_id"],
            "category": case["category"],
            "trigger": queued["trigger"],
            "review_target_hash": item["review_target_hash"],
            "environment_hash": environment["environment_hash"],
            "visible_chunk_ids": [row["chunk_id"] for row in environment["visible_chunks"]],
            "research_chunk_ids": [row["chunk_id"] for row in environment["research_chunks"]],
            "excluded_chunk_ids": [row["chunk_id"] for row in environment["excluded_chunks"]],
            "question": case["question"],
            "answer_key": case["answer_key"],
            "required_claim_ids": case["required_claim_ids"],
            "required_claims": case["required_claims"],
            "evidence_chunk_ids": case["evidence_chunk_ids"],
            "missing_claim_ids": case["missing_claim_ids"],
            "draft_status": drafts[case_id]["draft_status"],
            "review_status": drafts[case_id]["review_status"],
            "human_review_status": drafts[case_id]["human_review_status"],
            "model_review": reviews[case_id],
            "human_decision": None,
            "human_reason": None,
            "requested_changes": [],
        }
    return result


def build_expected_evidence_from_fixtures(ids: set[str]) -> dict[str, dict]:
    chunks = {
        row["chunk_id"]: row
        for row in load_jsonl(FIXTURES / "corpus/chunks.jsonl")
    }
    claims = {
        row["claim_id"]: row
        for row in load_jsonl(FIXTURES / "corpus/claims.jsonl")
    }
    return {
        chunk_id: {
            **chunks[chunk_id],
            "claims": [claims[claim_id] for claim_id in chunks[chunk_id]["claim_ids"]],
        }
        for chunk_id in sorted(ids)
    }


def test_cards_equal_current_fixture_projection() -> None:
    cards = parse_named_json_blocks(read(CHECKLIST), "review-card")
    expected = build_expected_cards_from_fixtures()
    assert cards == expected


def test_evidence_directory_is_complete_exact_and_deduplicated() -> None:
    cards = parse_named_json_blocks(read(CHECKLIST), "review-card")
    evidence = parse_named_json_blocks(read(CHECKLIST), "evidence-chunk")
    referenced = {
        chunk_id
        for card in cards.values()
        for field in ("visible_chunk_ids", "research_chunk_ids", "excluded_chunk_ids")
        for chunk_id in card[field]
    }
    assert set(evidence) == referenced
    assert evidence == build_expected_evidence_from_fixtures(referenced)
